import time import pandas as pd import numpy as np import MetaTrader5 as mt5 # --- ALL ACTIVE SYMBOLS --- SYMBOLS = ["EURUSD", "GBPUSD", "USDJPY", "XAUUSD", "BTCUSD.r"] TIMEFRAME = mt5.TIMEFRAME_M5 EMA_PERIOD = 50 RSI_PERIOD = 14 ATR_PERIOD = 14 RISK_PER_TRADE = 0.01 # 1% Risk framework def initialize_system(): """Safely connects to MT5 terminal and selects asset matrix.""" if not mt5.initialize(): print("❌ ERROR: MT5 open karo pehle! Connection failed.") quit() print("🔄 Syncing symbols with your broker broker infrastructure...") for s in SYMBOLS: # Check if symbol exists before adding info = mt5.symbol_info(s) if info is None: print(f"⚠️ Warning: Symbol {s} not found on this broker account. Skipping.") continue mt5.symbol_select(s, True) print(f"✅ Bot ON & Monitoring: {', '.join(SYMBOLS)}") def get_indicators(symbol): """Calculates smoothed indicator frames and isolates the last completed bar.""" bars = mt5.copy_rates_from_pos(symbol, TIMEFRAME, 0, 300) if bars is None or len(bars) < 100: return None df = pd.DataFrame(bars) # Trend calculation df['ema'] = df['close'].ewm(span=EMA_PERIOD, adjust=False).mean() # Momentum calculation delta = df['close'].diff() gain = (delta.where(delta > 0, 0)).rolling(window=RSI_PERIOD).mean() loss = (-delta.where(delta < 0, 0)).rolling(window=RSI_PERIOD).mean() rs = gain / (loss + 1e-10) df['rsi'] = 100 - (100 / (1 + rs)) # Volatility calculation hl = df['high'] - df['low'] hc = np.abs(df['high'] - df['close'].shift()) lc = np.abs(df['low'] - df['close'].shift()) df['atr'] = pd.concat([hl, hc, lc], axis=1).max(axis=1).rolling(window=ATR_PERIOD).mean() return df.iloc[-2] # Confirmed completely closed candle def get_filling_mode(symbol): """Dynamically fetches your broker's required filling mode to prevent order rejects.""" sym_info = mt5.symbol_info(symbol) if sym_info is None: return mt5.ORDER_FILLING_FOK filling = sym_info.filling_mode if filling & mt5.SYMBOL_FILLING_FOK: return mt5.ORDER_FILLING_FOK elif filling & mt5.SYMBOL_FILLING_IOC: return mt5.ORDER_FILLING_IOC return mt5.ORDER_FILLING_RETURN def execute_trade(symbol, order_type, calculated_atr): """Executes trades with auto-detected sizing and dynamic trailing risk brackets.""" tick = mt5.symbol_info_tick(symbol) if tick is None: print(f"❌ Failed to fetch live price ticks for {symbol}") return price = tick.ask if order_type == mt5.ORDER_TYPE_BUY else tick.bid # Dynamic Volatility Math Boundaries if order_type == mt5.ORDER_TYPE_BUY: stop_loss = price - (calculated_atr * 1.5) take_profit = price + (calculated_atr * 2.5) else: stop_loss = price + (calculated_atr * 1.5) take_profit = price - (calculated_atr * 2.5) # Dynamic Sizing Distribution Rules if "XAU" in symbol or "BTC" in symbol: lot = 0.01 # Safe minimum for volatile commodities/crypto else: lot = 0.10 # Standard baseline size for liquid Forex pairs # Automated Order Object Generation request = { "action": mt5.TRADE_ACTION_DEAL, "symbol": symbol, "volume": lot, "type": order_type, "price": round(price, 5), "sl": round(stop_loss, 5), "tp": round(take_profit, 5), "deviation": 20, "magic": 123456, "comment": "Multi Bot Engine", "type_time": mt5.ORDER_TIME_GTC, "type_filling": get_filling_mode(symbol) } result = mt5.order_send(request) if result is None: print(f"❌ Critical Execution Error on {symbol}: No response from server.") elif result.retcode != mt5.TRADE_RETCODE_DONE: print(f"❌ Order Rejected on {symbol}. Broker Error Code: {result.retcode}") else: print(f"🚀 Execution Successful! {symbol} Entry at {price:.5f}") def scan_all(): """Scans whole matrix arrays and enforces entry execution protocols.""" for symbol in SYMBOLS: # Check active open trades to prevent repeating position exposure positions = mt5.positions_get(symbol=symbol) if positions and len(positions) > 0: continue data_row = get_indicators(symbol) if data_row is None: continue current_close = data_row['close'] current_ema = data_row['ema'] current_rsi = data_row['rsi'] current_atr = data_row['atr'] print(f"[{symbol}] Price: {current_close:.5f} | RSI: {current_rsi:.1f} | EMA: {current_ema:.5f}") # --- REFINED MULTI-INDICATOR SYSTEM CONFIRMATION --- if current_close > current_ema and current_rsi < 35: print(f"🎯 SYSTEM CHECK: BUY SIGNAL TRIGGERED FOR {symbol}") execute_trade(symbol, mt5.ORDER_TYPE_BUY, current_atr) elif current_close < current_ema and current_rsi > 65: print(f"🎯 SYSTEM CHECK: SELL SIGNAL TRIGGERED FOR {symbol}") execute_trade(symbol, mt5.ORDER_TYPE_SELL, current_atr) # --- ENGINE EXECUTION LOOP --- initialize_system() try: while True: scan_all() print("--- Scanning finished. Sleeping for 60 seconds... ---\n") time.sleep(60) except KeyboardInterrupt: print("\n🛑 Execution paused and safety mechanisms disconnected.") finally: mt5.shutdown()
