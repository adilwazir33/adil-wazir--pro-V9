📋 TradingView Pine Script v5 (Copy karo) 
//@version=5
indicator("50 EMA + Stoch Scalper", overlay=true)
ema50 = ta.ema(close, 50)
plot(ema50, color=color.orange, linewidth=2)
k = ta.stoch(close, high, low, 5)
d = ta.sma(k, 3)
buy = close>ema50 and low<=ema50 and ta.crossover(k,d) and k[1]<20
sell = close=ema50 and ta.crossunder(k,d) and k[1]>80
plotshape(buy, style=shape.labelup, color=color.green, text="BUY")
plotshape(sell, style=shape.labeldown, color=color.red, text="SELL")
alertcondition(buy, title="BUY", message="BUY Pullback")
alertcondition(sell, title="SELL", message="SELL Pullback")
